6// The typeclass encoding under study, reduced to the fragment the model
7// covers. It mirrors the mechanism of beman.transpose (functor.hpp and
8// detail/typeclass_base.hpp in github.com/bemanproject/typeclass), which is
9// the reference implementation: a lookup variable template per typeclass
10// with std::false_type as the default instance, a derivation base that
11// supplies the derived operations from an instance's primitives, and an
12// operation object that resolves the instance at the call site through a
13// trailing defaulted constant template parameter.
14//
15// Each piece is here so that the emitted differential tests and the probes
16// can refer to one spelling, and so that the obligations table in
17// docs/plan.md has code to point at: soft failure of a missing instance is
18// the std::false_type default plus the constraint on the operation object.
19

27/** The canonical Functor lookup. Specialize for each type that has an
28 * instance; the primary is the default instance, std::false_type, so that
29 * a type without an instance is a soft failure at the use, never a hard
30 * error inside the lookup. */
31template <class T>
32inline constexpr auto functor = std::false_type{};
33
34/** True when a lookup has been specialized away from its default: the
35 * type has an instance. Pass decltype(functor<T>). */
36template <class LOOKUP_RESULT>
37inline constexpr bool has_instance_v =
38 !std::is_same_v<std::remove_cvref_t<LOOKUP_RESULT>, std::false_type>;
39
40/** The type has a Functor instance. */
41template <class T>
42concept has_functor = has_instance_v<decltype(functor<T>)>;

60/** Derivation base for Functor instances. `Impl` supplies `fmap(f, v)`;
61 * `replace` is derived from it unless `Impl` supplies its own. `Impl` is a
62 * protected base, as in beman.transpose, so the instance exposes only the
63 * operations this base declares. */
64template <class Impl>
65struct derive_functor : protected Impl {
66 template <class FUNCTION, class T>
67 constexpr auto fmap(FUNCTION &&function, T &&value) const
68 requires requires(const Impl &impl) {
69 impl.fmap(std::forward<FUNCTION>(function), std::forward<T>(value));
70 }
71 {
72 return Impl::fmap(std::forward<FUNCTION>(function),
73 std::forward<T>(value));
74 }
75
76 template <class T, class U>
77 constexpr auto replace(T &&value, U &&replacement) const
78 requires requires(const Impl &impl) {
79 impl.replace(std::forward<T>(value), std::forward<U>(replacement));
80 } || requires(const Impl &impl) {
81 impl.fmap(
82 detail::constant_function<std::remove_cvref_t<U>>{
83 std::forward<U>(replacement)},
84 std::forward<T>(value));
85 }
86 {
87 if constexpr (requires(const Impl &impl) {
88 impl.replace(std::forward<T>(value),
89 std::forward<U>(replacement));
90 }) {
91 return Impl::replace(std::forward<T>(value),
92 std::forward<U>(replacement));
93 } else {
94 return Impl::fmap(
95 detail::constant_function<std::remove_cvref_t<U>>{
96 std::forward<U>(replacement)},
97 std::forward<T>(value));
98 }
99 }
100};

103/** The operation object for `fmap`. The container type keys the lookup
104 * and is deduced from the second argument; the instance arrives through
105 * the trailing constant template parameter, a reference defaulted to the
106 * lookup. The constraint is the call itself, as in beman.transpose, so a
107 * type without an instance is not a candidate. The default is substituted
108 * before the constraint is checked ([temp.deduct.general]/5), which is the
109 * step-order hazard of the plan: it is safe here only because the lookup's
110 * default is std::false_type, which has no `fmap`. */
111inline constexpr struct fmap_t {
112 template <class FUNCTION, class T,
113 const auto &TC = functor<std::remove_cvref_t<T>>>
114 requires requires {
115 TC.fmap(std::declval<FUNCTION>(), std::declval<T>());
116 }
117 constexpr auto operator()(FUNCTION &&function, T &&value) const {
118 return TC.fmap(std::forward<FUNCTION>(function),
119 std::forward<T>(value));
120 }
121} fmap{};
